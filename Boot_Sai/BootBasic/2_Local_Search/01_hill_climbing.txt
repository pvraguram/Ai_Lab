# HILL CLIMBING - maximize any function f(x)
# Move to the best neighbour while it improves; stop at a peak.
# Run: python 01_hill_climbing.txt
import random

def f(x):
    return -x**2 + 10*x          # max at x = 5, f = 25 (change for any question)

def hill_climbing(x, step=0.1, max_iter=1000):
    for i in range(max_iter):
        neighbors = [x - step, x + step]
        best = max(neighbors, key=f)
        if f(best) <= f(x):      # no better neighbour -> peak reached
            break
        x = best
        print(f"iter {i}: x = {x:.2f}, f(x) = {f(x):.2f}")
    return x

start = random.uniform(-10, 10)
print("Start x =", round(start, 2))
x = hill_climbing(start)
print(f"Maximum at x = {x:.2f}, f(x) = {f(x):.2f}")
