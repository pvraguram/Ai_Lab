# DEPTH LIMITED SEARCH (DLS) and ITERATIVE DEEPENING DFS (IDDFS)
# DLS  : DFS that stops at a depth limit
# IDDFS: run DLS with limit 0, 1, 2, ... until goal is found
# Run: python 03_dls_iddfs.txt
graph = {
    'A': ['B', 'C'],
    'B': ['D', 'E'],
    'C': ['F', 'G'],
    'D': ['H'], 'E': [], 'F': [], 'G': ['I'], 'H': [], 'I': []
}

def dls(node, goal, limit, path):
    path = path + [node]
    if node == goal:
        return path
    if limit == 0:
        return None
    for nb in graph[node]:
        result = dls(nb, goal, limit - 1, path)
        if result:
            return result
    return None

def iddfs(start, goal, max_depth=10):
    for depth in range(max_depth + 1):
        print("Trying depth limit", depth)
        result = dls(start, goal, depth, [])
        if result:
            return result
    return None

print("DLS (limit 2):", dls('A', 'I', 2, []))
print("IDDFS path   :", iddfs('A', 'I'))
