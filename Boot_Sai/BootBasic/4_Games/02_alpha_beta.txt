# ALPHA-BETA PRUNING on a game tree (any depth)
# alpha = best for MAX so far, beta = best for MIN so far
# prune when alpha >= beta. Same answer as minimax, fewer nodes.
# Run: python 02_alpha_beta.txt
import math

tree = [
    [[3, 5], [6, 9]],
    [[1, 2], [0, -1]],
]
visited = []

def alphabeta(node, alpha, beta, is_max):
    if isinstance(node, int):
        visited.append(node)
        return node
    if is_max:
        value = -math.inf
        for child in node:
            value = max(value, alphabeta(child, alpha, beta, False))
            alpha = max(alpha, value)
            if alpha >= beta:
                print("Pruned at MAX node")
                break
        return value
    value = math.inf
    for child in node:
        value = min(value, alphabeta(child, alpha, beta, True))
        beta = min(beta, value)
        if alpha >= beta:
            print("Pruned at MIN node")
            break
    return value

print("Value:", alphabeta(tree, -math.inf, math.inf, True))
print("Leaves evaluated:", visited, "out of 8")
