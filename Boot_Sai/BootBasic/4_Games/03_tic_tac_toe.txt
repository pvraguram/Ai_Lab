# TIC-TAC-TOE - Human (O) vs AI (X) using MINIMAX
# Run: python 03_tic_tac_toe.txt   (enter cell 1-9)
board = [' '] * 9
WIN = [(0,1,2),(3,4,5),(6,7,8),(0,3,6),(1,4,7),(2,5,8),(0,4,8),(2,4,6)]

def winner(b):
    for a, c, d in WIN:
        if b[a] != ' ' and b[a] == b[c] == b[d]:
            return b[a]
    return 'Draw' if ' ' not in b else None

def minimax(b, is_max):
    w = winner(b)
    if w == 'X': return 1
    if w == 'O': return -1
    if w == 'Draw': return 0
    scores = []
    for i in range(9):
        if b[i] == ' ':
            b[i] = 'X' if is_max else 'O'
            scores.append(minimax(b, not is_max))
            b[i] = ' '
    return max(scores) if is_max else min(scores)

def ai_move():
    best, move = -2, None
    for i in range(9):
        if board[i] == ' ':
            board[i] = 'X'
            s = minimax(board, False)
            board[i] = ' '
            if s > best:
                best, move = s, i
    board[move] = 'X'

def show():
    for r in range(3):
        print(" | ".join(board[3*r:3*r+3]))
    print()

while winner(board) is None:
    show()
    try:
        i = int(input("Your move (1-9): ")) - 1
    except ValueError:
        print("Enter a number 1-9"); continue
    if not (0 <= i < 9) or board[i] != ' ':
        print("Invalid"); continue
    board[i] = 'O'
    if winner(board) is None:
        ai_move()
show()
print("Result:", winner(board))
