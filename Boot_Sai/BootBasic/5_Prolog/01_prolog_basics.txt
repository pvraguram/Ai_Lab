% ============================================================
% PROLOG BASICS - facts, rules, recursion, arithmetic, lists

% ============================================================

% ---------- FACTS ----------
parent(john, mary).
parent(mary, ann).
male(john).
female(mary). female(ann).

% ---------- RULES ----------
mother(X, Y)      :- parent(X, Y), female(X).
grandparent(X, Z) :- parent(X, Y), parent(Y, Z).

% ---------- RECURSION ----------
ancestor(X, Y) :- parent(X, Y).
ancestor(X, Y) :- parent(X, Z), ancestor(Z, Y).

% ---------- ARITHMETIC ----------
factorial(0, 1) :- !.
factorial(N, F) :- N > 0, N1 is N - 1, factorial(N1, F1), F is N * F1.

fib(0, 0) :- !.
fib(1, 1) :- !.
fib(N, F) :- N1 is N - 1, N2 is N - 2, fib(N1, F1), fib(N2, F2), F is F1 + F2.

% ---------- LISTS ----------
my_length([], 0).
my_length([_|T], N) :- my_length(T, N1), N is N1 + 1.

my_member(X, [X|_]).
my_member(X, [_|T]) :- my_member(X, T).

my_append([], L, L).
my_append([H|T], L, [H|R]) :- my_append(T, L, R).

my_reverse([], []).
my_reverse([H|T], R) :- my_reverse(T, RT), my_append(RT, [H], R).

sum_list_([], 0).
sum_list_([H|T], S) :- sum_list_(T, S1), S is S1 + H.

max_list_([X], X).
max_list_([H|T], M) :- max_list_(T, M1), M is max(H, M1).

% ---------- QUERIES ----------
% ?- mother(M, ann).            -> M = mary.
% ?- grandparent(john, G).      -> G = ann.
% ?- ancestor(john, X).         -> X = mary ; X = ann.
% ?- factorial(5, F).           -> F = 120.
% ?- fib(10, F).                -> F = 55.
% ?- my_length([a,b,c], N).     -> N = 3.
% ?- my_reverse([1,2,3], R).    -> R = [3,2,1].
% ?- my_append([1,2], [3], L).  -> L = [1,2,3].
% ?- sum_list_([1,2,3,4], S).   -> S = 10.
% ?- max_list_([3,9,2], M).     -> M = 9.
