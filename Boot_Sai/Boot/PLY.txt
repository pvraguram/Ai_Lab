# Each root move has possible opponent replies and their scores
game_tree = {
    "Move A": {"Reply 1": 3, "Reply 2": 5},
    "Move B": {"Reply 1": 2, "Reply 2": 9},
    "Move C": {"Reply 1": 4, "Reply 2": 6}
}

best_move = None
best_value = float("-inf")

for move, replies in game_tree.items():
    # MIN chooses the smallest score after this move
    value = min(replies.values())

    print(move, "-> opponent's best reply value:", value)

    # MAX chooses the move with the highest resulting value
    if value > best_value:
        best_value = value
        best_move = move

print("Best move:", best_move)
print("Minimax value:", best_value)
