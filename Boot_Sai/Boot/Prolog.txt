% ============================================================
% 5. FORWARD CHAINING
% ============================================================

% Initial facts

fc_fact(a).
fc_fact(b).


% Rules

fc_rule(a, c).
fc_rule(b, d).
fc_rule(c, e).
fc_rule(d, e).


% Get all initial facts

initial_facts(Facts) :-
    findall(X, fc_fact(X), Facts).


% Find facts that can be derived

derive(Facts, NewFacts) :-

    findall(
        Y,
        (
            fc_rule(X, Y),
            member(X, Facts),
            \+ member(Y, Facts)
        ),
        Temp
    ),

    sort(Temp, NewFacts).


% Stop when no new facts

forward_chain(Facts, Facts) :-
    derive(Facts, []),
    !.


% Continue deriving

forward_chain(Facts, FinalFacts) :-

    derive(Facts, NewFacts),

    append(Facts, NewFacts, UpdatedFacts),

    forward_chain(UpdatedFacts, FinalFacts).


% Run forward chaining

run_forward_chaining(Result) :-

    initial_facts(Facts),

    forward_chain(Facts, Result).

% ----------------------------
% Adjacent cells
% ----------------------------

adjacent(X, Y, X2, Y) :-
    X2 is X + 1.

adjacent(X, Y, X2, Y) :-
    X2 is X - 1.

adjacent(X, Y, X, Y2) :-
    Y2 is Y + 1.

adjacent(X, Y, X, Y2) :-
    Y2 is Y - 1.


% ----------------------------
% Breeze
% ----------------------------

% A cell has breeze if an
% adjacent cell contains a pit.

breeze(X, Y) :-
    adjacent(X, Y, A, B),
    pit(A, B).


% ----------------------------
% Stench
% ----------------------------

% A cell has stench if an
% adjacent cell contains Wumpus.

stench(X, Y) :-
    adjacent(X, Y, A, B),
    wumpus(A, B).


% ----------------------------
% Safe cell
% ----------------------------

safe(X, Y) :-
    \+ pit(X, Y),
    \+ wumpus(X, Y).

% ============================================================
% SEND + MORE = MONEY
% ============================================================

send_more_money(S, E, N, D, M, O, R, Y) :-

    Vars = [S,E,N,D,M,O,R,Y],

    Vars ins 0..9,

    all_distinct(Vars),

    S #\= 0,
    M #\= 0,

    1000*S + 100*E + 10*N + D
    +
    1000*M + 100*O + 10*R + E
    #=
    10000*M + 1000*O + 100*N + 10*E + Y,

    labeling([], Vars).

% ============================================================
%              RESOLUTION IN PROLOG
% ============================================================

% A clause is represented as a list of literals.
%
% p       = P
% not_p   = NOT P
%
% Example:
% [p, q]      means P OR Q
% [not_p]     means NOT P


% ============================================================
% COMPLEMENTARY LITERALS
% ============================================================

complement(p, not_p).
complement(not_p, p).

complement(q, not_q).
complement(not_q, q).

complement(r, not_r).
complement(not_r, r).


% ============================================================
% RESOLVE TWO CLAUSES
% ============================================================

resolve(C1, C2, Resolvent) :-

    member(L1, C1),

    complement(L1, L2),

    member(L2, C2),

    delete(C1, L1, R1),

    delete(C2, L2, R2),

    append(R1, R2, Temp),

    sort(Temp, Resolvent).


% ============================================================
% EXAMPLE
%
% C1 = P OR Q
% C2 = NOT P
%
% Result = Q
% ============================================================

example_resolution(Result) :-

    C1 = [p, q],

    C2 = [not_p],

    resolve(C1, C2, Result).


% Query:
%
% ?- example_resolution(R).
%
% R = [q].


CNF:


complement(p, not_p).
complement(not_p, p).
complement(q, not_q).
complement(not_q, q).

resolve(C1, C2, Result) :-
    member(L1, C1),
    complement(L1, L2),
    member(L2, C2),
    select(L1, C1, R1),
    select(L2, C2, R2),
    append(R1, R2, Temp),
    sort(Temp, Result).

% CNF clauses:
% P -> Q becomes NOT P OR Q
clause1([not_p, q]).
clause2([p]).
clause3([not_q]).

prove_q(Result) :-
    clause1(C1),
    clause2(C2),
    resolve(C1, C2, Derived),
    clause3(C3),
    resolve(Derived, C3, Result).
