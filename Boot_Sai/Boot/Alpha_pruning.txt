# Alpha-Beta Pruning

def alphabeta(board, depth, alpha, beta, is_maximizing):

    # Terminal states
    if board == ["X", "X", "X"]:
        return 10 - depth

    if board == ["O", "O", "O"]:
        return depth - 10

    if "_" not in board:
        return 0

    if is_maximizing:

        best_score = -float("inf")

        for i in range(len(board)):

            if board[i] == "_":

                board[i] = "X"

                score = alphabeta(
                    board,
                    depth + 1,
                    alpha,
                    beta,
                    False
                )

                board[i] = "_"

                best_score = max(best_score, score)

                alpha = max(alpha, best_score)

                # Pruning
                if beta <= alpha:
                    break

        return best_score

    else:

        best_score = float("inf")

        for i in range(len(board)):

            if board[i] == "_":

                board[i] = "O"

                score = alphabeta(
                    board,
                    depth + 1,
                    alpha,
                    beta,
                    True
                )

                board[i] = "_"

                best_score = min(best_score, score)

                beta = min(beta, best_score)

                # Pruning
                if beta <= alpha:
                    break

        return best_score


board = ["X", "O", "_"]

score = alphabeta(
    board,
    0,
    -float("inf"),
    float("inf"),
    True
)

print("Best score:", score)
