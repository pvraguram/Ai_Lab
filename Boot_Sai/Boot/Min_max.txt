AI = "X"
HUMAN = "O"

board = [[" " for _ in range(3)] for _ in range(3)]


def print_board():
    for row in board:
        print(" | ".join(row))
        print("-" * 9)


def winner(b):
    lines = []

    lines.extend(b)
    lines.extend([[b[r][c] for r in range(3)] for c in range(3)])
    lines.append([b[i][i] for i in range(3)])
    lines.append([b[i][2-i] for i in range(3)])

    for line in lines:
        if line[0] != " " and line.count(line[0]) == 3:
            return line[0]

    if all(b[r][c] != " " for r in range(3) for c in range(3)):
        return "draw"

    return None


def minimax(depth, maximizing):
    result = winner(board)

    if result == AI:
        return 10 - depth
    if result == HUMAN:
        return depth - 10
    if result == "draw":
        return 0

    if maximizing:
        best = float("-inf")

        for r in range(3):
            for c in range(3):
                if board[r][c] == " ":
                    board[r][c] = AI
                    score = minimax(depth + 1, False)
                    board[r][c] = " "
                    best = max(best, score)

        return best

    best = float("inf")

    for r in range(3):
        for c in range(3):
            if board[r][c] == " ":
                board[r][c] = HUMAN
                score = minimax(depth + 1, True)
                board[r][c] = " "
                best = min(best, score)

    return best


def best_move():
    best_score = float("-inf")
    move = None

    for r in range(3):
        for c in range(3):
            if board[r][c] == " ":
                board[r][c] = AI
                score = minimax(0, False)
                board[r][c] = " "

                if score > best_score:
                    best_score = score
                    move = (r, c)

    return move


while True:
    print_board()
    result = winner(board)

    if result:
        print("Draw!" if result == "draw" else result + " wins!")
        break

    # Human enters row and column from 1 to 3
    try:
        r, c = map(int, input("Enter row and column (1-3): ").split())
        r -= 1
        c -= 1

        if not (0 <= r < 3 and 0 <= c < 3 and board[r][c] == " "):
            print("Invalid move. Try again.")
            continue

        board[r][c] = HUMAN

    except ValueError:
        print("Enter two numbers, for example: 2 3")
        continue

    if winner(board):
        continue

    move = best_move()
    if move:
        board[move[0]][move[1]] = AI
